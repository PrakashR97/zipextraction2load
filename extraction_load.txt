public void Main()
{
    string folder =
        Dts.Variables["User::CsvRootFolder"]
        .Value.ToString();

    if (!System.IO.Directory.Exists(folder))
    {
        Dts.Events.FireError(
            0, "Month Folder",
            "Folder not found: " + folder,
            "", 0);

        Dts.TaskResult = (int)ScriptResults.Failure;
        return;
    }

    Dts.TaskResult = (int)ScriptResults.Success;
}

public void Main()
{
    try
    {
        string csvPath =
            Dts.Variables["User::CsvFilePath"]
            .Value.ToString();

        System.IO.FileInfo file =
            new System.IO.FileInfo(csvPath);

        if (!file.Exists)
            throw new System.IO.FileNotFoundException(
                "CSV not found", csvPath);

        Dts.Variables["User::CsvFileName"].Value =
            file.Name;

        Dts.Variables["User::CsvCategory"].Value =
            System.IO.Path.GetFileNameWithoutExtension(
                file.Name).Trim();

        Dts.Variables["User::SourceFolderName"].Value =
            file.Directory.Name;

        Dts.Variables["User::CsvProcessingID"].Value =
            System.Guid.NewGuid().ToString();

        Dts.TaskResult = (int)ScriptResults.Success;
    }
    catch (System.Exception ex)
    {
        Dts.Events.FireError(
            0, "Identify CSV",
            ex.ToString(), "", 0);

        Dts.TaskResult = (int)ScriptResults.Failure;
    }
}

CREATE OR REPLACE PROCEDURE FINALIZE_CSV_LOAD
(
    P_PROCESSING_ID IN VARCHAR2
)
AS
    V_CATEGORY     VARCHAR2(200);
    V_FOLDER       VARCHAR2(500);
    V_STAGE        VARCHAR2(128);
    V_TARGET       VARCHAR2(128);
    V_COLS         VARCHAR2(32767);
    V_SELECT_COLS  VARCHAR2(32767);
    V_SQL          VARCHAR2(32767);
    V_ROWS         NUMBER := 0;
    V_COUNT        NUMBER;
BEGIN
    SELECT CSV_CATEGORY, SOURCE_FOLDER_NAME
    INTO V_CATEGORY, V_FOLDER
    FROM CSV_LOAD_LOG
    WHERE PROCESSING_ID = P_PROCESSING_ID
      AND STATUS = 'PROCESSING'
    FOR UPDATE;

    -- Guard against a previously successful folder
    SELECT COUNT(*)
    INTO V_COUNT
    FROM CSV_LOAD_LOG
    WHERE SOURCE_FOLDER_NAME = V_FOLDER
      AND STATUS = 'SUCCESS';

    IF V_COUNT > 0 THEN
        RAISE_APPLICATION_ERROR(
            -20005,
            'Folder already loaded: ' || V_FOLDER
        );
    END IF;

    CASE UPPER(TRIM(V_CATEGORY))
        WHEN 'ACTIVITY LOG' THEN
            V_STAGE := 'STG_ACTIVITY_LOG';
            V_TARGET := 'ACTIVITY_LOG';
        WHEN 'AUDIT CLAIM FILE' THEN
            V_STAGE := 'STG_AUDIT_CLAIM_FILE';
            V_TARGET := 'AUDIT_CLAIM_FILE';
        WHEN 'AUDIT' THEN
            V_STAGE := 'STG_AUDIT';
            V_TARGET := 'AUDIT';
        WHEN 'CASE OVERVIEW TEST' THEN
            V_STAGE := 'STG_CASE_OVERVIEW_TEST';
            V_TARGET := 'CASE_OVERVIEW_TEST';
        WHEN 'DOCUMENTS' THEN
            V_STAGE := 'STG_DOCUMENTS';
            V_TARGET := 'DOCUMENTS';
        WHEN 'FINANCIALS' THEN
            V_STAGE := 'STG_FINANCIALS';
            V_TARGET := 'FINANCIALS';
        WHEN 'LEAD DETAILS DESCRIPTION' THEN
            V_STAGE := 'STG_LEAD_DETAILS_DESCRIPTION';
            V_TARGET := 'LEAD_DETAILS_DESCRIPTION';
        WHEN 'LINKED CASES' THEN
            V_STAGE := 'STG_LINKED_CASES';
            V_TARGET := 'LINKED_CASES';
        WHEN 'MANAGER' THEN
            V_STAGE := 'STG_MANAGER';
            V_TARGET := 'MANAGER';
        WHEN 'OVERVIEW' THEN
            V_STAGE := 'STG_OVERVIEW';
            V_TARGET := 'OVERVIEW';
        WHEN 'PARTICIPANT' THEN
            V_STAGE := 'STG_PARTICIPANT';
            V_TARGET := 'PARTICIPANT';
        WHEN 'REFERRAL' THEN
            V_STAGE := 'STG_REFERRAL';
            V_TARGET := 'REFERRAL';
        ELSE
            RAISE_APPLICATION_ERROR(
                -20002, 'Unsupported CSV category'
            );
    END CASE;

    -- Verify metadata columns
    SELECT COUNT(*)
    INTO V_COUNT
    FROM USER_TAB_COLUMNS
    WHERE (TABLE_NAME = V_STAGE
           AND COLUMN_NAME = 'PROCESSING_ID')
       OR (TABLE_NAME = V_TARGET
           AND COLUMN_NAME IN
               ('SOURCE_FOLDER_NAME',
                'LOAD_SYSTEMTIMESTAMP'));

    IF V_COUNT <> 3 THEN
        RAISE_APPLICATION_ERROR(
            -20003, 'Required metadata missing'
        );
    END IF;

    -- Build common business column mappings
    FOR R IN (
        SELECT S.COLUMN_NAME
        FROM USER_TAB_COLUMNS S
        JOIN USER_TAB_COLUMNS T
          ON T.TABLE_NAME = V_TARGET
         AND T.COLUMN_NAME = S.COLUMN_NAME
        WHERE S.TABLE_NAME = V_STAGE
          AND S.COLUMN_NAME NOT IN
              ('PROCESSING_ID',
               'SOURCE_FOLDER_NAME',
               'LOAD_SYSTEMTIMESTAMP')
          AND S.VIRTUAL_COLUMN = 'NO'
          AND T.VIRTUAL_COLUMN = 'NO'
          AND T.IDENTITY_COLUMN = 'NO'
        ORDER BY S.COLUMN_ID
    )
    LOOP
        IF V_COLS IS NOT NULL THEN
            V_COLS := V_COLS || ',';
            V_SELECT_COLS := V_SELECT_COLS || ',';
        END IF;

        V_COLS := V_COLS ||
            DBMS_ASSERT.ENQUOTE_NAME(
                R.COLUMN_NAME, FALSE);

        V_SELECT_COLS := V_SELECT_COLS ||
            'S.' || DBMS_ASSERT.ENQUOTE_NAME(
                R.COLUMN_NAME, FALSE);
    END LOOP;

    IF V_COLS IS NULL THEN
        RAISE_APPLICATION_ERROR(
            -20004, 'No matching business columns'
        );
    END IF;

    V_SQL :=
        'INSERT INTO ' ||
        DBMS_ASSERT.ENQUOTE_NAME(V_TARGET, FALSE) ||
        ' (' || V_COLS || ', SOURCE_FOLDER_NAME)' ||
        ' SELECT ' || V_SELECT_COLS || ', :1' ||
        ' FROM ' ||
        DBMS_ASSERT.ENQUOTE_NAME(V_STAGE, FALSE) ||
        ' S WHERE S.PROCESSING_ID = :2';

    EXECUTE IMMEDIATE V_SQL
    USING V_FOLDER, P_PROCESSING_ID;

    V_ROWS := SQL%ROWCOUNT;

    UPDATE CSV_LOAD_LOG
    SET STATUS = 'SUCCESS',
        ROWS_LOADED = V_ROWS,
        END_TIME = SYSTIMESTAMP,
        ERROR_MESSAGE = NULL
    WHERE PROCESSING_ID = P_PROCESSING_ID;

    COMMIT;

EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        RAISE;
END FINALIZE_CSV_LOAD;
/





BEGIN
    FINALIZE_CSV_LOAD(?);
END;




UPDATE CSV_LOAD_LOG
SET STATUS = 'FAILED',
    END_TIME = SYSTIMESTAMP,
    ERROR_MESSAGE =
        'CSV load failed. Check SSIS execution logs.'
WHERE PROCESSING_ID = ?
  AND STATUS = 'PROCESSING'
